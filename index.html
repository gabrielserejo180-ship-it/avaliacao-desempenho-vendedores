/** * SISTEMA DE AVALIAÇÃO DE DESEMPENHO DE VENDEDORES — BACKEND * ------------------------------------------------------------ * Como publicar: * 1. Crie uma planilha Google nova. * 2. Extensões > Apps Script, cole este arquivo como Code.gs. * 3. Rode a função configurarPlanilha() uma vez (menu ▶ ao lado dela) * para criar as abas "Vendedores" e "Avaliacoes" com os cabeçalhos. * 4. Implantar > Nova implantação > Tipo: App da Web. * - Executar como: Eu * - Quem pode acessar: Qualquer pessoa * 5. Copie a URL do Web App e cole em CONFIG.API_URL no index.html. */ const ABA_VENDEDORES = "Vendedores"; const ABA_AVALIACOES = "Avaliacoes"; const INDICADORES = [ "satisfacaoGeral", "recomendacao", "recompra", "infraestrutura", "atendimento", "entrega", "negociacao", "testeRide" ]; function configurarPlanilha(){ const ss = SpreadsheetApp.getActiveSpreadsheet(); let abaVendedores = ss.getSheetByName(ABA_VENDEDORES); if(!abaVendedores){ abaVendedores = ss.insertSheet(ABA_VENDEDORES); abaVendedores.appendRow(["ID", "CPF", "Nome", "Cargo", "FotoURL", "TotalManual", "MediaManual"]); } let abaAvaliacoes = ss.getSheetByName(ABA_AVALIACOES); if(!abaAvaliacoes){ abaAvaliacoes = ss.insertSheet(ABA_AVALIACOES); abaAvaliacoes.appendRow([ "ID", "VendedorID", "Data", "SatisfacaoGeral", "Recomendacao", "Recompra", "Infraestrutura", "Atendimento", "Entrega", "Negociacao", "TesteRide" ]); } } function doPost(e){ try{ const body = JSON.parse(e.postData.contents); const action = body.action; let resultado; if(action === "buscarVendedor"){ resultado = buscarVendedor_(body.termo, body.manual); } else if(action === "salvarFoto"){ resultado = salvarFoto_(body.vendedorId, body.foto); } else if(action === "novaAvaliacao"){ resultado = novaAvaliacao_(body.vendedorId, body.notas); } else if(action === "importarVendedores"){ resultado = importarVendedores_(body.lista); } else { resultado = { erro: "Ação desconhecida: " + action }; } return ContentService.createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); }catch(err){ return ContentService.createTextOutput(JSON.stringify({ erro: err.message + " | " + err.stack })) .setMimeType(ContentService.MimeType.JSON); } } function apenasDigitos_(s){ return (s || "").toString().replace(/\D/g, ""); } function ehCpf_(s){ return apenasDigitos_(s).length === 11; } function idParaTermo_(termo){ const digitos = apenasDigitos_(termo); return ehCpf_(termo) ? digitos : ("nome:" + termo.trim().toLowerCase()); } function lerVendedores_(){ const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_VENDEDORES); const dados = sheet.getDataRange().getValues(); const linhas = dados.slice(1); return linhas.map((linha, i) => ({ linha: i + 2, id: linha[0], cpf: linha[1], nome: linha[2], cargo: linha[3], foto: linha[4], totalManual: linha[5] === "" || linha[5] === undefined ? null : Number(linha[5]), mediaManual: linha[6] === "" || linha[6] === undefined ? null : Number(linha[6]), })); } function buscarVendedor_(termo, manual){ const vendedores = lerVendedores_(); let vendedor = null; let novo = false; if(ehCpf_(termo)){ const cpf = apenasDigitos_(termo); vendedor = vendedores.find(v => apenasDigitos_(v.cpf) === cpf) || null; } else { const alvo = termo.trim().toLowerCase(); vendedor = vendedores.find(v => (v.nome || "").toLowerCase().indexOf(alvo) !== -1) || null; } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_VENDEDORES); if(!vendedor){ novo = true; const id = idParaTermo_(termo); const cpf = ehCpf_(termo) ? apenasDigitos_(termo) : ""; const nome = termo.trim(); const novaLinha = sheet.getLastRow() + 1; sheet.appendRow([id, cpf, nome, "Consultor de Vendas", "", "", ""]); sheet.getRange(novaLinha, 1).setNumberFormat("@").setValue(id); vendedor = { id, cpf, nome, cargo: "Consultor de Vendas", foto: "", totalManual: null, mediaManual: null, linha: novaLinha }; } if(manual && (manual.total !== null && manual.total !== undefined || manual.media !== null && manual.media !== undefined)){ if(manual.total !== null && manual.total !== undefined){ vendedor.totalManual = manual.total; sheet.getRange(vendedor.linha || acharLinhaVendedor_(sheet, vendedor.id), 6).setValue(manual.total); } if(manual.media !== null && manual.media !== undefined){ vendedor.mediaManual = manual.media; sheet.getRange(vendedor.linha || acharLinhaVendedor_(sheet, vendedor.id), 7).setValue(manual.media); } } const resumo = calcularResumo_(vendedor); return { vendedor: vendedor, resumo: resumo, novo: novo }; } function acharLinhaVendedor_(sheet, vendedorId){ const dados = sheet.getDataRange().getValues(); for(let i = 1; i < dados.length; i++){ if(String(dados[i][0]) === String(vendedorId)) return i + 1; } return -1; } function calcularResumo_(vendedor){ const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_AVALIACOES); const dados = sheet.getDataRange().getValues(); const idAlvo = String(vendedor.id); const linhas = dados.slice(1).filter(l => String(l[1]) === idAlvo); const total = linhas.length; const medias = {}; INDICADORES.forEach((key, idx) => { if(total === 0){ medias[key] = null; return; } const soma = linhas.reduce((acc, l) => acc + (Number(l[3 + idx]) || 0), 0); medias[key] = soma / total; }); let mediaGeral = null; if(total > 0){ const somaGeral = INDICADORES.reduce((acc, key) => acc + (medias[key] || 0), 0); mediaGeral = somaGeral / INDICADORES.length; } return { total: total, mediaGeral: mediaGeral, medias: medias, manual: false }; } function salvarFoto_(vendedorId, fotoDataUrl){ // Salva a foto (já comprimida pelo front-end) direto como texto na planilha, // em vez de subir pro Drive — evita os problemas de CORS/hotlink do Drive // tanto na exibição quanto na geração da imagem para compartilhar. if(fotoDataUrl.length > 49000){ throw new Error("Imagem grande demais para salvar na planilha (" + fotoDataUrl.length + " caracteres)."); } const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_VENDEDORES); const dados = sheet.getDataRange().getValues(); let linha = -1; for(let i = 1; i < dados.length; i++){ if(String(dados[i][0]) === String(vendedorId)){ linha = i + 1; break; } } if(linha === -1) throw new Error("Vendedor não encontrado: " + vendedorId); sheet.getRange(linha, 5).setValue(fotoDataUrl); return { ok: true }; } function importarVendedores_(lista){ const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_VENDEDORES); const vendedores = lerVendedores_(); const cpfsExistentes = {}; const nomesExistentes = {}; vendedores.forEach(v => { if(v.cpf) cpfsExistentes[apenasDigitos_(v.cpf)] = true; nomesExistentes[(v.nome || "").trim().toLowerCase()] = true; }); let importados = 0, duplicados = 0, invalidos = 0; const novasLinhas = []; (lista || []).forEach(item => { const nome = (item.nome || "").trim(); const cpf = apenasDigitos_(item.cpf || ""); if(!nome){ invalidos++; return; } const chave = cpf.length === 11 ? cpf : nome.toLowerCase(); const jaExiste = cpf.length === 11 ? cpfsExistentes[cpf] : nomesExistentes[chave]; if(jaExiste){ duplicados++; return; } const id = cpf.length === 11 ? cpf : ("nome:" + nome.toLowerCase()); novasLinhas.push([id, cpf, nome, "Consultor de Vendas", "", "", ""]); if(cpf.length === 11) cpfsExistentes[cpf] = true; nomesExistentes[chave] = true; importados++; }); if(novasLinhas.length > 0){ sheet.getRange(sheet.getLastRow() + 1, 1, novasLinhas.length, 7).setValues(novasLinhas); } return { importados: importados, duplicados: duplicados, invalidos: invalidos }; } function novaAvaliacao_(vendedorId, notas){ const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(ABA_AVALIACOES); const id = Utilities.getUuid(); const linha = sheet.getLastRow() + 1; sheet.appendRow([ id, String(vendedorId), new Date(), notas.satisfacaoGeral || 0, notas.recomendacao || 0, notas.recompra || 0, notas.infraestrutura || 0, notas.atendimento || 0, notas.entrega || 0, notas.negociacao || 0, notas.testeRide || 0 ]); // força a coluna VendedorID como texto, para não virar número e desalinhar com o ID do vendedor sheet.getRange(linha, 2).setNumberFormat("@").setValue(String(vendedorId)); return { ok: true }; }
